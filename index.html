My Java Project

CC102 - Daily Sales Analyzer

This section contains my Java console-based game developed as part of my CC102 Final Performance Task. 


public class DailySalesAnalyzer {
    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);

        double[] sales = new double[7];
        double total = 0;
        double average;
        int count = 0;

        for (int i = 0; i < 7; i++) {
            while (true) {
                System.out.print("Enter sales for Day " + (i + 1) + ": ");

                if (input.hasNextDouble()) {
                    sales[i] = input.nextDouble();

                    if (sales[i] >= 0) {
                        break;
                    } else {
                        System.out.println("Invalid amount!");
                    }
                } else {
                    System.out.println("Invalid input!");
                    input.next();
                }
            }

            total += sales[i];

            if (sales[i] >= 5000) {
                count++;
            }
        }

        average = total / 7;

        double highest = sales[0];
        double lowest = sales[0];

        for (int i = 1; i < 7; i++) {
            if (sales[i] > highest) {
                highest = sales[i];
            }

            if (sales[i] < lowest) {
                lowest = sales[i];
            }
        }

        for (int i = 0; i < 6; i++) {
            for (int j = 0; j < 6 - i; j++) {
                if (sales[j] < sales[j + 1]) {
                    double temp = sales[j];
                    sales[j] = sales[j + 1];
                    sales[j + 1] = temp;
                }
            }
        }

        System.out.println("DAILY SALES REPORT");
        System.out.println("Total Sales: PHP " + total);
        System.out.println("Average Sales: PHP " + average);
        System.out.println("Highest Sales: PHP " + highest);
        System.out.println("Lowest Sales: PHP " + lowest);
        System.out.println("Days with at least PHP 5,000: " + count);

        System.out.println("Sales in Descending Order:");

        for (int i = 0; i < 7; i++) {
            System.out.println(sales[i]);
        }

        input.close();
    }
}
